**目标**：将前端同步阻塞式 AI 问答改为 SSE 流式接收，实现真正的"打字机"视觉效果——AI 每输出一个字，前端立刻显示，无需等待完整回答。

## 步骤 1/5: 认知学习 — EventSource 与 ReadableStream（前 10 分钟，不写代码）

### 为什么不能用原来的 `fetch + JSON`

```javascript
// 原来的同步模式
const result = await fetchAPI('/api/documents/1/ask', {
    method: 'POST',
    body: JSON.stringify({ question })
});
// ↑ 这里会阻塞，等服务器返回完整 JSON 后才继续执行
```

流式模式下，服务器不会等全部内容生成完再返回，而是**边生成边推送**。前端需要**边接收边显示**。

### 两种 SSE 客户端方案

| 方案 | API | 支持自定义 Header | 适用场景 |
|:---|:---|:---|:---|
| **EventSource** | 浏览器原生 | ❌ 不支持 | 公开接口（无需 Token）|
| **fetch + ReadableStream** | 浏览器原生 | ✅ 支持 | 需要 Authorization 的接口 |

本项目所有业务接口都需要 JWT Token（`Authorization: Bearer xxx`），所以**不能用原生 EventSource**，必须用 `fetch + ReadableStream`。

### fetch + ReadableStream 的工作原理

```
前端                                         服务器
  │ ───── GET /ask/stream ────────────────→  │
  │ ←──── Response (stream) ───────────────  │
  │                                          │ AI 生成 "S"
  │ ←──── chunk: data: S ──────────────────  │
  │         ↑ 前端立刻显示 "S"                │ AI 生成 "p"
  │ ←──── chunk: data: p ──────────────────  │
  │         ↑ 前端立刻显示 "p"                │ AI 生成 "r"
  │ ←──── chunk: data: r ──────────────────  │
  │         ↑ 前端立刻显示 "r"                │ ...
  │ ←──── done ────────────────────────────  │
```

### SSE 数据格式解析

服务器推送的原始数据长这样：

```
data: Spring
data: Boot
data: 是

data: 一个
```

每行以 `data:` 开头，`\n\n`（两个换行）表示一条消息结束。前端需要：
1. 用 `TextDecoder` 把二进制 `Uint8Array` 转成字符串
2. 按行分割，提取 `data:` 后面的内容
3. 逐段回调给 UI 更新

## 步骤 2/5: 新增流式 API 调用函数（核心改动）

在 `app.js` 中新增一套流式请求封装，**保留原有同步函数不动**（向后兼容）。

### 2.1 通用 SSE 流式请求封装

在 `app.js` 的 `fetchAPI` 函数**下方**添加：

```javascript
/**
 * 通用 SSE 流式请求封装（使用 fetch + ReadableStream）
 * @param {string} url - API 路径（不含 base）
 * @param {Object} params - URL 查询参数（会被序列化为 ?key=value）
 * @param {Function} onChunk - 每收到一个 token 时的回调函数
 * @param {Function} onError - 发生错误时的回调函数（可选）
 */
async function fetchStream(url, params, onChunk, onError) {
    // 拼查询参数
    const queryString = new URLSearchParams(params).toString();
    const fullUrl = `${API_BASE}${url}${queryString ? '?' + queryString : ''}`;

    // 构造请求头（带上 JWT Token）
    const headers = {};
    if (authToken) {
        headers['Authorization'] = 'Bearer ' + authToken;
    }

    try {
        const response = await fetch(fullUrl, { headers });

        if (!response.ok) {
            throw new Error(`请求失败：${response.status} ${response.statusText}`);
        }

        const reader = response.body.getReader();
        const decoder = new TextDecoder();
        let buffer = '';  // 缓存不完整的行

        while (true) {
            const { done, value } = await reader.read();
            if (done) break;

            // 把二进制 chunk 解码成字符串
            buffer += decoder.decode(value, { stream: true });

            // 按行分割，处理完整的 SSE 行
            const lines = buffer.split('\n');
            buffer = lines.pop();  // 最后一行可能不完整，保留到下次

            for (const line of lines) {
                if (line.startsWith('data:')) {
                    const data = line.slice(5).trim();
                    if (data) {
                        onChunk(data);
                    }
                }
            }
        }

        // 处理 buffer 中剩余的内容
        if (buffer.startsWith('data:')) {
            const data = buffer.slice(5).trim();
            if (data) onChunk(data);
        }

    } catch (error) {
        if (onError) onError(error);
        else throw error;
    }
}
```

### 2.2 文档流式问答 API

```javascript
/**
 * 文档级流式问答 API
 * @param {number} documentId - 文档 ID
 * @param {string} question - 用户问题
 * @param {boolean} useDocumentContext - 是否基于文档回答
 * @param {Function} onChunk - 每收到一个 token 的回调
 */
function askDocumentStreamAPI(documentId, question, useDocumentContext, onChunk) {
    return fetchStream(
        `/api/documents/${documentId}/ask/stream`,
        { question, useDocumentContext },
        onChunk
    );
}
```

### 2.3 笔记本流式问答 API

```javascript
/**
 * 笔记本级流式问答 API
 * @param {number} notebookId - 笔记本 ID
 * @param {string} question - 用户问题
 * @param {Function} onChunk - 每收到一个 token 的回调
 */
function askNotebookStreamAPI(notebookId, question, onChunk) {
    return fetchStream(
        `/api/notebooks/${notebookId}/ask/stream`,
        { question },
        onChunk
    );
}
```

            <span class="stream-dot"></span> AI 正在输入...

    </div>

</div>

### 3.2 修改 JS — 接入 EventSource

把 `app.js` 中的 `askDocument()` 函数**整个替换**：

```javascript
async function askDocument() {
    const input = document.getElementById('qaInput');
    const question = input.value.trim();
    const currentDocId = document.getElementById('viewDocumentTitle').dataset.documentId;
    const useDocumentContext = document.getElementById('qaContextSwitch').checked;

    if (!question) {
        showToast('请输入问题', 'warning');
        return;
    }

    // 获取 DOM 元素
    const answerWrapper = document.getElementById('qaAnswer');
    const answerEl = document.getElementById('qaAnswerText');
    const indicator = document.getElementById('qaStreamIndicator');

    // 显示回答区，清空内容，显示"输入中"指示器
    answerWrapper.style.display = 'block';
    answerEl.value = '';
    indicator.style.display = 'inline';
    answerEl.classList.add('streaming');

    showToast('AI 正在思考...', 'info');

    try {
        await askDocumentStreamAPI(
            currentDocId,
            question,
            useDocumentContext,
            (chunk) => {
                // 每收到一个 token，追加到 textarea
                answerEl.value += chunk;
                // 自动滚动到底部
                answerEl.scrollTop = answerEl.scrollHeight;
            }
        );

        showToast('回答完成', 'success');
    } catch (error) {
        showToast('回答失败：' + error.message, 'error');
        answerEl.value = '获取回答失败，请稍后重试。';
    } finally {
        // 隐藏指示器，移除流式样式
        indicator.style.display = 'none';
        answerEl.classList.remove('streaming');
    }
}
```
